# Scenario + rollout environment for Linux/CUDA hosts.
#
# This installs the *client* side only: LIBERO simulation, the scenario code,
# and the openpi websocket client. The pi05_libero model itself is served by a
# separate process (scripts/serve_pi05.sh) using openpi's own uv environment,
# because openpi's JAX pins conflict with robosuite's.  Keeping them apart is
# what allows the renderer and the model to sit on different hosts.
#
#   conda create --prefix .conda/envs/ipu python=3.10 -y
#   conda activate ./.conda/envs/ipu
#   python -m pip install -r requirements-linux-gpu.txt
#   python -m pip install -e third_party/LIBERO
#   python -m pip install -e ../openpi/packages/openpi-client
#   python -m pip install -e ../Interaction-Uncertainty
